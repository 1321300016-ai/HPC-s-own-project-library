```c
 * 原: main.h 中写 “int raw_sock_fd;”（每个包含 main.h 的 .c 都定义一份）
 * 现: 定义移到 main.c 唯一一处，main.h 中只留 extern 声明

 *       达到上限时直接丢包，保证进程稳定。
 * ========================================================= */

#define MAX_IP_THREADS 16
/* ==================== 修改点（新增） ====================
 * 讲解: arp_cleanup() 原来从未被调用，ARP 表只增不减、无上限；
 *       主循环每收到 ARP_CLEANUP_INTERVAL 个包就清理一次超时表项。
 * ========================================================= */

int main(int argc, char const *argv[])
{

     *     if (get_net_interfaces() != 0) { ... } // 第 2 次调用，重复枚举
     * 现: if (get_net_interfaces() != 0) { ... } // 只调用一次并检查
     * 讲解: 原代码重复枚举网卡，第 1 次结果被直接丢弃；单次调用即可。
     * ========================================================= */
    if (get_net_interfaces() != 0) {

        exit(1);   /* 修改：exit(-1) → exit(1)，-1 作为退出码会被截断成 255 */

    for (int i = 0; i < g_iface_count; i++) {
        LOG_INFO("  %s: %d.%d.%d.%d",

                 g_ifaces[i].ip[0], g_ifaces[i].ip[1],
                 g_ifaces[i].ip[2], g_ifaces[i].ip[3]);
    }

    //键盘线程读取命令
    /*&KEY_T —— 线程标识符
    * NULL —— 线程属性,这里的 NULL：表示使用默认属性。也就是说，系统会按默认方式创建线程（通常是可合并的、默认栈大小）

    * NULL —— 传递给线程函数的参数

    pthread_create(&KEY_T, NULL, key_pthread, NULL);
    pthread_detach(KEY_T);

    * htons(ETH_P_ALL) —— 协议类型，ETH_P_ALL：表示接收所有类型的以太网帧
    */
    /* ==================== 修改点（新增） ====================
     * 讲解: 初始化转发线程限流信号量（进程内使用，pshared=0）。
     * ========================================================= */
    sem_init(&g_thread_sem, 0, MAX_IP_THREADS);

    raw_sock_fd = socket(PF_PACKET, SOCK_RAW, htons(ETH_P_ALL));
    if (raw_sock_fd < 0) {
        LOG_ERROR("socket创建失败");
        perror("socket");
        exit(1);   /* 修改：exit(-1) → exit(1)，-1 作为退出码会被截断成 255 */
    }
    char recv_buf[RECV_SIZE] = "";
    ssize_t recv_len = 0; //ssize_t —— 有符号整型
    //打印内存池状态
    int used, total, free_blocks;
    pool_stats(&g_packet_pool, &used, &total, &free_blocks);
    LOG_INFO("内存池初始化完成: 总块数=%d, 块大小=%d, 空闲=%d",

    LOG_INFO("进入主循环，开始收包...");
    unsigned long packet_counter = 0;   /* 修改点（新增）：用于周期触发 ARP 清理 */
    while(1){

        /*raw_sock_fd,套接字文件描述符

            LOG_ERROR("recvfrom 接收数据异常");
            perror("recvfrom");
            continue;

            /* 修改点（新增）：ARP 包最小长度校验（14 字节帧头 + 28 字节 ARP 头） */
            if (recv_len < 42) {
                LOG_WARN("ARP 包长度异常: %zd", recv_len);
                continue;
            }
            unsigned char ip[4];
            unsigned char mac[6];
            memcpy(mac, recv_buf + 22, 6);
            memcpy(ip, recv_buf + 28, 4);
            arp_update(ip, mac);
        }
        else if(eth_type == 0x0800){
            /* ==================== 修改点（原 → 现） ====================
             * 原: unsigned char *dst_ip = (unsigned char *)recv_buf + 30;
             *     // 30 = 14(以太网头) + 16(IP头内目的IP偏移)
             *     无长度校验、不读取 IHL，IP 头带选项时会取错位置
             * 现: 先校验长度与 IP 版本，再按 IHL 计算目的 IP 偏移
             * 讲解: 目的 IP 在 IP 头内偏移 16 字节；无选项时 IHL=20，
             *       帧内偏移 = 14 + 16 = 30。IP 头带选项时 IHL 更大，
             *       偏移 = 14 + 16 + (IHL*4 - 20)。
             * ========================================================= */
            if (recv_len < 34) {
                LOG_WARN("IP 包长度异常: %zd", recv_len);
                continue;
            }
            unsigned char ip_ver_ihl = recv_buf[14];
            if ((ip_ver_ihl >> 4) != 4) {
                LOG_DEBUG("非 IPv4 包，忽略");
                continue;
            }
            int ip_hdr_len = (ip_ver_ihl & 0x0F) * 4;
            unsigned char *dst_ip = (unsigned char *)recv_buf + 14 + 16 + (ip_hdr_len - 20);
            if(ip_filter_check(dst_ip)){
                LOG_DEBUG("IP %d.%d.%d.%d 在黑名单中，丢弃",
                        dst_ip[0], dst_ip[1], dst_ip[2], dst_ip[3]);
                continue;
            }
            RECV_DATA *packet = (RECV_DATA *)pool_alloc(&g_packet_pool);
            if(packet== NULL){
                LOG_ERROR("内存池满，丢弃包");
                continue;
            }
            /* ==================== 修改点（新增） ====================
             * 讲解: 转发线程限流——sem_trywait 成功才占用一个槽位，
             *       并发线程达到 MAX_IP_THREADS 时直接丢包（不阻塞收包循环）；
             *       线程结束后在 ip_pthread 里 sem_post 归还槽位。
             * ========================================================= */
            if (sem_trywait(&g_thread_sem) != 0) {
                LOG_WARN("转发线程数已达上限，丢包");
                pool_free(&g_packet_pool, packet);
                continue;
            }
            LOG_DEBUG("pool_alloc 成功: packet=%p", packet);   /* 修改：INFO → DEBUG，避免每个包刷屏 */
            packet->data_len = recv_len;
            memcpy(packet->data, recv_buf, recv_len);
            pthread_t IP_T;
            /* 修改点（新增）：检查 pthread_create 返回值，失败时归还内存与信号量槽位 */
            if (pthread_create(&IP_T, NULL, ip_pthread, packet) != 0) {
                LOG_ERROR("pthread_create 失败，丢包");
                pool_free(&g_packet_pool, packet);
                sem_post(&g_thread_sem);
                continue;
            }
            pthread_detach(IP_T);
        }
        else{
            LOG_DEBUG("非 IP/ARP 包，已忽略");
        }
        /* 修改点（新增）：周期性清理超时 ARP 表项（原代码从未调用 arp_cleanup） */
        if (++packet_counter % ARP_CLEANUP_INTERVAL == 0) {
            arp_cleanup();
        }
    }
    LOG_INFO("主循环退出，开始清理...");
    close(raw_sock_fd);
    ip_filter_save();
    ip_filter_destroy();
    arp_destroy();
    /* ==================== 修改点（原 → 现） ====================
     * 原: pool_stats(&g_iface_count, &used, &total, &free_blocks);  // 参数类型错误
     *     pool_reset(&g_iface_count);
     * 现: pool_stats(&g_packet_pool, ...);
     *     pool_reset(&g_packet_pool);
     * 讲解: 原代码把 int 变量的地址当 STATIC_POOL* 传入，类型完全不匹配；
     *       修正为真正的内存池对象。注意本段位于 while(1) 之后，
     *       正常运行时不可达——正常退出应走 key_pthread 的 exit 命令
     *       （exit_handler 中已保存黑名单并清理）。
     * ========================================================= */
    pool_stats(&g_packet_pool, &used, &total, &free_blocks);
    pool_reset(&g_packet_pool);
    LOG_INFO("内存池已重置");
    LOG_INFO("路由器已安全退出");

    return 0;
}
```

## arp_cache.c

```c
/**
 * arp_cache.c - ARP缓存实现
 */

#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <time.h>
#include <errno.h>

#include "arp_cache.h"
#include "logger.h"
#include <pthread.h>   /* 修改点（新增）：ARP 链表互斥锁 */

/* ==================== 全局变量 ==================== */

LIST_HEAD g_arp_head;

/* ==================== 修改点（新增） ====================
 * 讲解: ARP 链表由主线程（arp_update）、转发线程（arp_lookup）、
 *       键盘线程（arp_show）并发访问，必须加互斥锁保护；
 *       g_arp_target_ip 的赋值与链表遍历也在同一把锁内，
 *       避免“线程A设目标、线程B覆盖目标”的错配问题。
 * ========================================================= */
pthread_mutex_t g_arp_lock = PTHREAD_MUTEX_INITIALIZER;

/* ★★★ 用于 list_find 传递目标IP的全局变量 ★★★ */
static unsigned char *g_arp_target_ip = NULL;

/* ==================== 内部函数（给 simple_list 用） ==================== */

/**
 * ARP匹配函数：比较IP是否相同
 * 注意：list_find 只传一个参数，目标IP通过全局变量 g_arp_target_ip 传递
 */
static int arp_match(void *data)
{
    ARP_DATA *d = (ARP_DATA *)data;
    if (g_arp_target_ip == NULL) {
        return 1;  // 没有目标IP，不匹配
    }
    return (memcmp(d->ip, g_arp_target_ip, 4) == 0) ? 0 : 1;
}

/**
 * ARP打印函数（给 list_foreach 用）
 */
static void arp_print(void *data)
{
    ARP_DATA *d = (ARP_DATA *)data;
    time_t now = time(NULL);
    printf("%d.%d.%d.%d -> %02x:%02x:%02x:%02x:%02x:%02x  (age: %lds)\n",
           d->ip[0], d->ip[1], d->ip[2], d->ip[3],
           d->mac[0], d->mac[1], d->mac[2], d->mac[3], d->mac[4], d->mac[5],
           now - d->last_seen);
}

/**
 * ARP释放函数（给 list_destroy 用）
 */
static void arp_free(void *data)
{
    free(data);
}

/* ==================== 公共函数实现 ==================== */

void arp_update(unsigned char *ip, unsigned char *mac)
{
    LIST_NODE *node;
    ARP_DATA *data;
    
    if (ip == NULL || mac == NULL) {
        return;
    }
    
    /* ==================== 修改点（新增） ====================
     * 讲解: 加锁保护 ARP 链表。原代码没有任何同步：主线程 arp_update()
     *       与转发线程 arp_lookup()/arp_cleanup() 并发访问同一链表，
     *       会产生数据竞争甚至崩溃。
     * ========================================================= */
    pthread_mutex_lock(&g_arp_lock);
    
    /* 查找是否已存在 */
    g_arp_target_ip = ip;
    node = list_find(&g_arp_head, arp_match);
    
    if (node != NULL) {
        /* 已存在：更新MAC和时间戳 */
        data = (ARP_DATA *)node->data;
        memcpy(data->mac, mac, 6);
        data->last_seen = time(NULL);
        LOG_DEBUG("更新ARP: %d.%d.%d.%d -> %02x:%02x:%02x:%02x:%02x:%02x",
                  ip[0], ip[1], ip[2], ip[3],
                  mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
        pthread_mutex_unlock(&g_arp_lock);   /* 修改点（新增）：解锁后返回 */
        return;
    }
    
    /* 不存在：创建新表项 */
    data = (ARP_DATA *)malloc(sizeof(ARP_DATA));
    if (data == NULL) {
        LOG_ERROR("ARP缓存malloc失败: %s", strerror(errno));
        pthread_mutex_unlock(&g_arp_lock);   /* 修改点（新增）：失败分支也要解锁 */
        return;
    }
    
    memcpy(data->ip, ip, 4);
    memcpy(data->mac, mac, 6);
    data->last_seen = time(NULL);
    
    list_add_head(&g_arp_head, data);
    LOG_DEBUG("新增ARP: %d.%d.%d.%d -> %02x:%02x:%02x:%02x:%02x:%02x",
              ip[0], ip[1], ip[2], ip[3],
              mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
    pthread_mutex_unlock(&g_arp_lock);   /* 修改点（新增）：解锁后返回 */
}

/* ==================== 修改点（原 → 现） ====================
 * 原: unsigned char *arp_lookup(unsigned char *ip) —— 返回链表内部指针
 * 现: int arp_lookup(unsigned char *ip, unsigned char *out_mac)
 *     —— 锁内把 MAC 拷贝到调用者缓冲区，返回 0=命中 / -1=未命中
 * 讲解: 原接口暴露内部指针，锁释放后该节点可能被 arp_update/
 *       arp_cleanup 修改或释放，产生 use-after-free；改为拷贝返回值。
 * ========================================================= */
int arp_lookup(unsigned char *ip, unsigned char *out_mac)
{
    LIST_NODE *node;
    ARP_DATA *data;
    
    if (ip == NULL || out_mac == NULL) {
        return -1;
    }
    
    pthread_mutex_lock(&g_arp_lock);   /* 修改点（新增）：加锁保护链表 */
    g_arp_target_ip = ip;
    node = list_find(&g_arp_head, arp_match);
    
    if (node == NULL) {
        pthread_mutex_unlock(&g_arp_lock);
        return -1;
    }
    
    data = (ARP_DATA *)node->data;
    /* 更新访问时间（LRU思想） */
    data->last_seen = time(NULL);
    memcpy(out_mac, data->mac, 6);     /* 修改点：改为在锁内拷贝 MAC */
    pthread_mutex_unlock(&g_arp_lock);
    
    return 0;
}

void arp_cleanup(void)
{
    LIST_NODE *p, *next;
    ARP_DATA *data;
    time_t now = time(NULL);
    int removed = 0;
    
    /* ==================== 修改点（新增） ====================
     * 讲解: 加锁保护 ARP 链表（原代码无锁，与 arp_update/arp_lookup 并发）。
     * ========================================================= */
    pthread_mutex_lock(&g_arp_lock);
    
    p = g_arp_head.first;
    while (p != NULL) {
        next = p->next;
        data = (ARP_DATA *)p->data;
        
        if (now - data->last_seen > ARP_TIMEOUT) {
            /* 超时，删除该节点 */
            list_delete_node(&g_arp_head, p, arp_free);
            removed++;
        }
        p = next;
    }
    
    if (removed > 0) {
        LOG_DEBUG("ARP清理: 删除了 %d 条超时表项，剩余 %d 条",
                  removed, g_arp_head.count);
    }
    pthread_mutex_unlock(&g_arp_lock);   /* 修改点（新增）：清理结束解锁 */
}

void arp_show(void)
{
    /* ==================== 修改点（新增） ====================
     * 讲解: 加锁保护 ARP 链表（原代码无锁，show 由键盘线程调用，
     *       与主线程/转发线程并发读写同一链表）。
     * ========================================================= */
    pthread_mutex_lock(&g_arp_lock);
    
    if (g_arp_head.count == 0) {
        printf("\nARP缓存为空\n");
        pthread_mutex_unlock(&g_arp_lock);
        return;
    }
    
    printf("\n========== ARP缓存 ==========\n");
    printf("IP              MAC                          Age(s)\n");
    printf("----------------------------------------------\n");
    list_foreach(&g_arp_head, arp_print);
    printf("----------------------------------------------\n");
    printf("总数: %d\n", g_arp_head.count);
    printf("=============================\n\n");
    pthread_mutex_unlock(&g_arp_lock);   /* 修改点（新增）：打印结束解锁 */
}

void arp_destroy(void)
{
    /* ==================== 修改点（新增） ====================
     * 讲解: 加锁后再释放整条链表，避免与其它线程的并发访问冲突。
     * ========================================================= */
    pthread_mutex_lock(&g_arp_lock);
    list_destroy(&g_arp_head, arp_free);
    pthread_mutex_unlock(&g_arp_lock);
}
```

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <arpa/inet.h>
#include <errno.h>
#include <pthread.h>   /* 修改点（新增）：黑名单链表互斥锁 */

#include "ip_filter.h"
#include "logger.h"

#define IP_CONFIG_FILE "ip_config"

LIST_HEAD g_ip_filter_head;

unsigned char *g_target_ip = NULL;

/* ==================== 修改点（新增） ====================
 * 讲解: 黑名单链表由主线程（ip_filter_check）与键盘线程
 *       （ip_filter_add/del/show）并发访问，必须加互斥锁；
 *       g_target_ip 的赋值与链表遍历也在同一把锁内，
 *       避免线程间互相覆盖目标 IP。
 * ========================================================= */
pthread_mutex_t g_ip_filter_lock = PTHREAD_MUTEX_INITIALIZER;

/* ==================== 比较和打印函数（给 simple_list 用） ==================== */

/**
 * IP比较函数（符合 list_find 要求：参数是 void*）
 */
static int ip_match(void *data)
{
    IP_DATA *d = (IP_DATA *)data;
    // 外部目标IP通过全局变量传递（因为list_find只传一个参数）
    extern unsigned char *g_target_ip;
    /* ==================== 修改点（原 → 现） ====================
     * 原: return (memcmp(d->ip, g_target_ip, 4) == 0) ? 0 : 1;
     *     —— 若调用方忘记先设置 g_target_ip（值为 NULL），
     *        memcmp 会访问空指针 → 段错误
     * 现: 先判断 g_target_ip 是否为 NULL，为空直接不匹配
     * ========================================================= */
    if (g_target_ip == NULL) {
        return 1;  // 没有目标IP，不匹配
    }
    return (memcmp(d->ip, g_target_ip, 4) == 0) ? 0 : 1;
}

/**
 * IP打印函数（给 list_foreach 用）
 */
static void ip_print(void *data)
{
    IP_DATA *d = (IP_DATA *)data;
    printf("%d.%d.%d.%d\n", d->ip[0], d->ip[1], d->ip[2], d->ip[3]);
}

/**
 * IP释放函数（给 list_destroy 用）
 */
static void ip_free(void *data)
{
    free(data);
}

/* ==================== 公共函数实现 ==================== */

void ip_filter_init(void)
{
    FILE *fp;
    char line[64];
    
    list_init(&g_ip_filter_head);
    
    fp = fopen(IP_CONFIG_FILE, "r");
    if (fp == NULL) {
        LOG_WARN("IP配置文件不存在，跳过加载");
        return;
    }
    
    while (fgets(line, sizeof(line), fp) != NULL) {
        unsigned char ip[4];
        struct in_addr addr;   /* 修改点：原为 unsigned int raw_ip，改为语义明确的 in_addr */
        
        /* 去掉换行符 */
        line[strcspn(line, "\n")] = '\0';
        /* 去掉末尾的\r（Windows换行） */
        line[strcspn(line, "\r")] = '\0';
        
        /* 跳过空行 */
        if (strlen(line) == 0) {
            continue;
        }
        
        if (inet_pton(AF_INET, line, &addr) <= 0) {
            LOG_WARN("无效IP格式: %s", line);
            continue;
        }
        memcpy(ip, &addr, 4);
        
        ip_filter_add(ip);
    }
    
    fclose(fp);
    LOG_INFO("IP黑名单加载完成，共 %d 条", g_ip_filter_head.count);
}

void ip_filter_save(void)
{
    FILE *fp;
    LIST_NODE *p;
    
    fp = fopen(IP_CONFIG_FILE, "w");
    if (fp == NULL) {
        LOG_ERROR("无法打开配置文件写入: %s", strerror(errno));
        return;
    }
    
    /* 修改点（新增）：加锁遍历，避免保存期间链表被其它线程修改 */
    pthread_mutex_lock(&g_ip_filter_lock);
    p = g_ip_filter_head.first;
    while (p != NULL) {
        IP_DATA *d = (IP_DATA *)p->data;
        fprintf(fp, "%d.%d.%d.%d\n", d->ip[0], d->ip[1], d->ip[2], d->ip[3]);
        p = p->next;
    }
    pthread_mutex_unlock(&g_ip_filter_lock);
    
    fclose(fp);
    LOG_INFO("IP黑名单已保存，共 %d 条", g_ip_filter_head.count);
}

int ip_filter_add(unsigned char *ip)
{
    IP_DATA *data;
    LIST_NODE *node;
    
    /* ==================== 修改点（原 → 现） ====================
     * 原: 直接 list_find(&g_ip_filter_head, ip_match);
     *     —— 没有先给 g_target_ip 赋值，ip_match() 里 memcmp 空指针，
     *        只要黑名单非空（配置文件 ≥2 行）就段错误！
     * 现: 先 g_target_ip = ip 绑定目标，再加锁执行查找与插入
     * 讲解: ip_match() 依赖全局 g_target_ip 做比较；加锁保证
     *       g_target_ip 的赋值与链表操作原子，不被其它线程覆盖。
     * ========================================================= */
    pthread_mutex_lock(&g_ip_filter_lock);
    
    g_target_ip = ip;
    node = list_find(&g_ip_filter_head, ip_match);
    if (node != NULL) {
        LOG_DEBUG("IP %d.%d.%d.%d 已在黑名单中", ip[0], ip[1], ip[2], ip[3]);
        pthread_mutex_unlock(&g_ip_filter_lock);   /* 修改点（新增）：失败也要解锁 */
        return -1;
    }
    
    data = (IP_DATA *)malloc(sizeof(IP_DATA));
    if (data == NULL) {
        LOG_ERROR("malloc失败: %s", strerror(errno));
        pthread_mutex_unlock(&g_ip_filter_lock);   /* 修改点（新增）：失败也要解锁 */
        return -1;
    }
    memcpy(data->ip, ip, 4);
    
    list_add_head(&g_ip_filter_head, data);
    LOG_DEBUG("添加IP到黑名单: %d.%d.%d.%d", ip[0], ip[1], ip[2], ip[3]);
    pthread_mutex_unlock(&g_ip_filter_lock);   /* 修改点（新增）：解锁后返回 */
    return 0;
}

int ip_filter_del(unsigned char *ip)
{
    void *deleted;
    
    /* ==================== 修改点（原 → 现） ====================
     * 原: 直接 list_delete(&g_ip_filter_head, ip_match);
     *     —— 与 ip_filter_add 同样的空指针问题，delip 命令一执行就崩
     * 现: 先设置 g_target_ip，再加锁执行删除
     * ========================================================= */
    pthread_mutex_lock(&g_ip_filter_lock);
    
    g_target_ip = ip;
    deleted = list_delete(&g_ip_filter_head, ip_match);
    if (deleted == NULL) {
        LOG_DEBUG("IP %d.%d.%d.%d 不在黑名单中", ip[0], ip[1], ip[2], ip[3]);
        pthread_mutex_unlock(&g_ip_filter_lock);   /* 修改点（新增）：失败也要解锁 */
        return -1;
    }
    
    free(deleted);
    LOG_DEBUG("从黑名单删除IP: %d.%d.%d.%d", ip[0], ip[1], ip[2], ip[3]);
    pthread_mutex_unlock(&g_ip_filter_lock);   /* 修改点（新增）：解锁后返回 */
    return 0;
}

int ip_filter_check(unsigned char *ip)
{
    LIST_NODE *node;
    
    /* ★★★ 关键：使用全局变量传递目标IP ★★★ */
    extern unsigned char *g_target_ip;
    /* 修改点（新增）：加锁，避免与键盘线程的 add/del 并发访问链表 */
    pthread_mutex_lock(&g_ip_filter_lock);
    g_target_ip = ip;
    node = list_find(&g_ip_filter_head, ip_match);
    pthread_mutex_unlock(&g_ip_filter_lock);
    return (node != NULL) ? 1 : 0;
}

void ip_filter_show(void)
{
    /* 修改点（新增）：加锁遍历，避免与主线程的 check 并发 */
    pthread_mutex_lock(&g_ip_filter_lock);
    
    if (g_ip_filter_head.count == 0) {
        printf("IP黑名单为空\n");
        pthread_mutex_unlock(&g_ip_filter_lock);
        return;
    }
    printf("\n========== IP黑名单 ==========\n");
    list_foreach(&g_ip_filter_head, ip_print);
    printf("总数: %d\n", g_ip_filter_head.count);
    printf("==============================\n\n");
    pthread_mutex_unlock(&g_ip_filter_lock);
}

void ip_filter_destroy(void)
{
    /* 修改点（新增）：加锁后释放整条链表 */
    pthread_mutex_lock(&g_ip_filter_lock);
    list_destroy(&g_ip_filter_head, ip_free);
    pthread_mutex_unlock(&g_ip_filter_lock);
}
```

## ip_pthread.c

```c
/**
 * ip_pthread.c - IP包转发线程
 * 
 * 这是整个路由器的核心模块！
 * 流程：
 *   1. 提取目的IP
 *   2. 查路由表找出口网卡
 *   3. 查ARP缓存获取目的MAC
 *   4. 没有则发ARP请求（最多3次）
 *   5. 修改数据包的MAC地址
 *   6. 从出口网卡转发出去
 */

#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <unistd.h>
#include <pthread.h>
#include <sys/socket.h>
#include <sys/ioctl.h>
#include <net/if.h>
#include <netinet/ether.h>
#include <netpacket/packet.h>
#include <errno.h>
#include <semaphore.h>   /* 修改点（新增）：转发线程限流信号量 */

#include "ip_pthread.h"
#include "arp_cache.h"
#include "net_iface.h"
#include "logger.h"
#include "static_pool.h"
#include <arpa/inet.h>
#include "main.h"


/* ==================== 外部引用 ==================== */

/* ==================== 修改点（原 → 现） ====================
 * 原: extern int raw_sock_fd;   // 本文件重复声明
 * 现: 删除本文件的 extern（main.h 中已提供 extern 声明）
 * 讲解: main.h 已统一声明 raw_sock_fd，这里再写一遍属于冗余，
 *       容易在修改时产生不一致。
 * ========================================================= */

/** 包内存池（在 main.c 中定义） */
extern STATIC_POOL g_packet_pool;

/** 转发线程限流信号量（在 main.c 中定义）—— 修改点（新增） */
extern sem_t g_thread_sem;


/* ==================== ARP请求包模板 ==================== */

/**
 * 42字节ARP请求包（以太网头 + ARP头）
 * 偏移说明：
 *   [0-5]   目的MAC（广播：FF:FF:FF:FF:FF:FF）
 *   [6-11]  源MAC（待填充）
 *   [12-13] 协议类型（0x0806 = ARP）
 *   [14-15] 硬件类型（0x0001 = Ethernet）
 *   [16-17] 协议类型（0x0800 = IP）
 *   [18]    硬件地址长度（6）
 *   [19]    协议地址长度（4）
 *   [20-21] 操作码（0x0001 = ARP请求）
 *   [22-27] 发送方MAC（待填充）
 *   [28-31] 发送方IP（待填充）
 *   [32-37] 目标MAC（待填充）
 *   [38-41] 目标IP（待填充）
 */
/* ==================== 修改点（原 → 现） ====================
 * 原: static unsigned char arp_buf[42] = {...};  // 全局静态缓冲区
 * 现: 不再使用全局缓冲区，ARP 请求包改为线程局部构造
 * 讲解: 多个转发线程会并发调用 create_arp_request()，若共用全局
 *       arp_buf，线程A填入的 IP/MAC 可能被线程B覆盖，发出错误请求。
 *       每个线程在自己的栈上构造缓冲区即可避免该竞态。
 * ========================================================= */


/* ==================== 私有函数 ==================== */

/**
 * 查找目的IP对应的出口网卡索引
 * 比较IP地址的前3字节（/24网段）
 * @param dst_ip  目的IP地址（4字节）
 * @return        网卡索引，如果找不到返回 g_iface_count
 */
static int find_network_segment(unsigned char *dst_ip)
{
    int i, j;
    
    for (i = 0; i < g_iface_count; i++) {
        /* ==================== 修改点（原 → 现） ====================
         * 原: 比较 IP 前 3 字节，硬编码假设所有网卡都是 /24
         * 现: 按每块网卡自己的 netmask 做网络号比较，并跳过 DOWN 接口
         * 讲解: 网卡可能是 /16、/8 等；只比前 3 字节会误判。
         *       (IP & mask) == (网卡IP & mask) 才是正确的同网段判断。
         * ========================================================= */
        if (!g_ifaces[i].is_up) {
            continue;   /* 跳过 DOWN 的网卡 */
        }
        for (j = 0; j < 4; j++) {
            if ((dst_ip[j] & g_ifaces[i].netmask[j]) !=
                (g_ifaces[i].ip[j] & g_ifaces[i].netmask[j])) {
                break;
            }
        }
        if (j == 4) {
            return i;   /* 4 个字节的网络号都匹配 */
        }
    }
    return g_iface_count;  // 未找到
}

/**
 * 从指定网卡发送数据
 * @param eth_name  网卡名称（如 "ens33"）
 * @param data      要发送的数据
 * @param data_len  数据长度
 * @return          实际发送字节数，-1=失败
 */
static int send_msg(const char *eth_name, unsigned char *data, unsigned long data_len)
{
    struct ifreq ethreq;
    struct sockaddr_ll sll;
    int ret;
    
    /* 获取网卡接口索引 */
    strncpy(ethreq.ifr_name, eth_name, IFNAMSIZ);
    if (ioctl(raw_sock_fd, SIOCGIFINDEX, &ethreq) == -1) {
        LOG_ERROR("ioctl SIOCGIFINDEX 失败: %s", strerror(errno));
        perror("ioctl");
        return -1;
    }
    
    /* 构造套接字地址 */
    memset(&sll, 0, sizeof(sll));
    sll.sll_ifindex = ethreq.ifr_ifindex;
    sll.sll_family = AF_PACKET;
    sll.sll_protocol = htons(ETH_P_ALL);
    
    /* 发送数据 */
    ret = sendto(raw_sock_fd, data, data_len, 0, 
                 (struct sockaddr *)&sll, sizeof(sll));
    if (ret < 0) {
        LOG_ERROR("sendto 失败: %s", strerror(errno));
        perror("sendto");
    }
    
    return ret;
}

/**
 * 组建ARP请求包
 * @param dst_ip        目的IP地址
 * @param iface_idx     出口网卡索引
 */
/* ==================== 修改点（原 → 现） ====================
 * 原: static void create_arp_request(unsigned char *dst_ip, int iface_idx)
 *     —— 操作全局 arp_buf
 * 现: static void create_arp_request(unsigned char *arp_buf, ...)
 *     —— 操作调用者传入的线程局部缓冲区
 * 讲解: 见上方全局 arp_buf 的说明；缓冲区由调用线程在栈上提供，
 *       不再跨线程共享。
 * ========================================================= */
static void create_arp_request(unsigned char *arp_buf, unsigned char *dst_ip, int iface_idx)
{
    NET_IFACE *iface = &g_ifaces[iface_idx];
    
    /* 从零开始构造 42 字节 ARP 请求 */
    memset(arp_buf, 0, 42);
    memset(arp_buf, 0xFF, 6);                  /* 目的MAC：广播 */
    arp_buf[12] = 0x08; arp_buf[13] = 0x06;   /* 协议类型：ARP */
    arp_buf[14] = 0x00; arp_buf[15] = 0x01;   /* 硬件类型：Ethernet */
    arp_buf[16] = 0x08; arp_buf[17] = 0x00;   /* 协议类型：IP */
    arp_buf[18] = 0x06;                       /* 硬件地址长度：6 */
    arp_buf[19] = 0x04;                       /* 协议地址长度：4 */
    arp_buf[20] = 0x00; arp_buf[21] = 0x01;   /* 操作码：ARP请求 */
    
    /* 填充源MAC */
    memcpy(arp_buf + 6, iface->mac, 6);
    memcpy(arp_buf + 22, iface->mac, 6);
    
    /* 填充源IP */
    memcpy(arp_buf + 28, iface->ip, 4);
    
    /* 填充目的IP */
    memcpy(arp_buf + 38, dst_ip, 4);
    
    /* 目的MAC保持全0（ARP请求中该字段为0，memset 后已满足） */
}


/* ==================== 线程入口函数 ==================== */

void *ip_pthread(void *arg)
{
    RECV_DATA *msg_data = (RECV_DATA *)arg;
    unsigned char *dst_ip;
    unsigned char mac[6];        /* 修改点：原为指针，现为局部数组（安全拷贝） */
    int iface_idx;
    int ret;
    int retry;
    
    if (msg_data == NULL) {
        return NULL;
    }
    
    /* ==================== 修改点（原 → 现） ====================
     * 原: dst_ip = (unsigned char *)msg_data->data + 30;  // 固定偏移，无校验
     * 现: 先校验长度与 IP 版本，再按 IHL 计算目的 IP 偏移
     * 讲解: 与 main.c 的解析保持一致；IP 头带选项时 IHL 更大，
     *       固定偏移 30 会取错位置。最小长度 34 = 14(帧头)+20(IP头)。
     * ========================================================= */
    if (msg_data->data_len < 34) {
        LOG_WARN("转发线程: IP 包长度异常: %zd", msg_data->data_len);
        goto IP_PTHREAD_END;
    }
    {
        unsigned char ver_ihl = msg_data->data[14];
        if ((ver_ihl >> 4) != 4) {
            LOG_DEBUG("转发线程: 非 IPv4 包，忽略");
            goto IP_PTHREAD_END;
        }
        int ip_hdr_len = (ver_ihl & 0x0F) * 4;
        dst_ip = (unsigned char *)msg_data->data + 14 + 16 + (ip_hdr_len - 20);
    }
    
    LOG_DEBUG("转发线程: 目的IP=%d.%d.%d.%d, 数据长度=%zd",
              dst_ip[0], dst_ip[1], dst_ip[2], dst_ip[3], 
              msg_data->data_len);
    
    /* 2. 查找出口网卡 */
    iface_idx = find_network_segment(dst_ip);
    if (iface_idx == g_iface_count) {
        LOG_WARN("无路由: %d.%d.%d.%d", 
                 dst_ip[0], dst_ip[1], dst_ip[2], dst_ip[3]);
        goto IP_PTHREAD_END;
    }
    LOG_DEBUG("出口网卡: %s", g_ifaces[iface_idx].name);
    
    /* 3. 检查广播地址（主机位全为1）—— 修改点：按 netmask 判断，不再只看第4字节 */
    {
        int j, is_bcast = 1;
        for (j = 0; j < 4; j++) {
            if ((dst_ip[j] | g_ifaces[iface_idx].netmask[j]) != 0xFF) {
                is_bcast = 0;
                break;
            }
        }
        if (is_bcast) {
            LOG_DEBUG("广播包，不转发");
            goto IP_PTHREAD_END;
        }
    }
    
    /* 4. 检查环回接口 */
    if (strcmp(g_ifaces[iface_idx].name, "lo") == 0) {
        LOG_DEBUG("环回接口，不转发");
        goto IP_PTHREAD_END;
    }
    
    /* 5. 查ARP缓存（修改点：改为安全拷贝接口，结果存入局部 mac[]） */
    if (arp_lookup(dst_ip, mac) != 0) {
        int mac_found = 0;   /* 修改点（新增）：记录重试期间是否命中 */
        
        /* 6. 如果没有MAC，发送ARP请求（最多3次） */
        LOG_DEBUG("ARP缓存未命中,发送ARP请求: %d.%d.%d.%d",
                  dst_ip[0], dst_ip[1], dst_ip[2], dst_ip[3]);
        
        for (retry = 0; retry < 3; retry++) {
            unsigned char arp_buf[42];   /* 修改点（新增）：线程局部缓冲区，替代原全局 arp_buf */
            
            /* 组建并发送ARP请求 */
            create_arp_request(arp_buf, dst_ip, iface_idx);
            ret = send_msg(g_ifaces[iface_idx].name, arp_buf, sizeof(arp_buf));
            if (ret < 0) {
                LOG_ERROR("发送ARP请求失败 (重试%d)", retry + 1);
            }
            
            /* 等待200ms */
            usleep(200000);
            
            /* 重新查询ARP缓存 */
            if (arp_lookup(dst_ip, mac) == 0) {
                mac_found = 1;
                LOG_DEBUG("ARP获取成功: %d.%d.%d.%d -> %02x:%02x:%02x:%02x:%02x:%02x",
                          dst_ip[0], dst_ip[1], dst_ip[2], dst_ip[3],
                          mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
                break;
            }
        }
        
        /* 3次都失败，放弃转发 */
        if (!mac_found) {
            LOG_WARN("ARP超时: %d.%d.%d.%d", 
                     dst_ip[0], dst_ip[1], dst_ip[2], dst_ip[3]);
            goto IP_PTHREAD_END;
        }
    }
    
    /* 7. 修改数据包的MAC地址（核心操作） */
    /*    目的MAC = 目标设备的MAC */
    memcpy(msg_data->data, mac, 6);
    /*    源MAC = 出口网卡的MAC */
    memcpy(msg_data->data + 6, g_ifaces[iface_idx].mac, 6);
    
    LOG_DEBUG("MAC已修改: 目的MAC=%02x:%02x:%02x:%02x:%02x:%02x",
              mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
    
    /* 8. 转发数据 */
    ret = send_msg(g_ifaces[iface_idx].name, 
                   msg_data->data, msg_data->data_len);
    if (ret < 0) {
        LOG_ERROR("转发失败: %s", strerror(errno));
    } else {
        LOG_DEBUG("转发成功: %d字节 -> %s", ret, g_ifaces[iface_idx].name);
    }

IP_PTHREAD_END:
    /* 9. 释放内存池块（★必须！★） */
    LOG_DEBUG("准备释放内存池块: msg_data=%p", msg_data);   /* 修改：INFO → DEBUG，避免每个包刷屏 */
    pool_free(&g_packet_pool, msg_data);
    /* 10. 归还转发线程限流信号量槽位（与 main.c 的 sem_trywait 对应）—— 修改点（新增） */
    sem_post(&g_thread_sem);
    LOG_DEBUG("转发线程结束，内存池块已释放");
    
    return NULL;
}
```

```c
/**
 * =============================================================================
 * 文件: cmd_parser.c
 * 功能: 实现 cmd_parser.h 中声明的所有函数
 * =============================================================================
 * 
 * 【内部数据结构】
 *   static CMD_ENTRY cmd_table[MAX_CMD]
 *   数组存储所有已注册的命令
 * 
 * 【调用关系】
 *   主程序初始化
 *        ↓
 *   cmd_register("setip", handler)  → 填充 cmd_table[]
 *   
 *   主程序键盘循环
 *        ↓
 *   cmd_execute("setip:192.168.1.1")
 *        ↓
 *   sscanf() 拆分为 cmd="setip", arg="192.168.1.1"
 *        ↓
 *   遍历 cmd_table[] 查找匹配
 *        ↓
 *   调用 handler(arg)
 * 
 *   cmd_show_help()
 *        ↓
 *   遍历 cmd_table[] 打印所有命令名
 * 
 * 【安全说明】
 *   - 命令名和参数都有长度限制，防止溢出
 * =============================================================================
 */

#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include "cmd_parser.h"

/* ==================== 内部常量 ==================== */

/** 最大支持的命令数量 */
#define MAX_CMD 64

/** 命令名最大长度（含结尾'\0'） */
#define CMD_NAME_LEN 64

/** 参数最大长度（含结尾'\0'） */
#define ARG_LEN 128

/* ==================== 内部数据结构 ==================== */

/**
 * 命令条目（存储命令名和处理函数的映射）
 */
typedef struct {
    char name[CMD_NAME_LEN];    /**< 命令名称 */
    cmd_handler_t handler;      /**< 处理函数指针 */
} CMD_ENTRY;

/* ==================== 内部全局变量 ==================== */

/** 命令表 */
static CMD_ENTRY cmd_table[MAX_CMD];

/** 当前已注册的命令数量 */
static int cmd_count = 0;

/* ==================== 函数实现 ==================== */

int cmd_register(const char *cmd, cmd_handler_t handler)
{
    int i;
    
    /* 检查参数有效性 */
    if (cmd == NULL || handler == NULL) {
        return -1;
    }
    
    /* 检查命令名是否包含 ':'（非法字符） */
    if (strchr(cmd, ':') != NULL) {
        return -1;
    }
    
    /* 检查是否已存在同名命令，如果存在则覆盖 */
    for (i = 0; i < cmd_count; i++) {
        if (strcmp(cmd_table[i].name, cmd) == 0) {
            cmd_table[i].handler = handler;
            return 0;
        }
    }
    
    /* 新增命令 */
    if (cmd_count >= MAX_CMD) {
        return -1;
    }
    
    /* ==================== 修改点（原 → 现） ====================
     * 原: strcpy(cmd_table[cmd_count].name, cmd);
     *     —— 命令名长度无校验，超长输入会写穿 name[64]
     * 现: 先检查长度，超长命令名直接拒绝
     * 讲解: 命令名来自用户输入，必须限制长度防止缓冲区溢出。
     * ========================================================= */
    if (strlen(cmd) >= CMD_NAME_LEN) {
        return -1;   /* 命令名过长 */
    }
    strcpy(cmd_table[cmd_count].name, cmd);
    cmd_table[cmd_count].handler = handler;
    cmd_count++;
    return 0;
}

int cmd_execute(char *input)
{
    char cmd[CMD_NAME_LEN] = "";
    char arg[ARG_LEN] = "";
    int i;
    
    if (input == NULL || input[0] == '\0') {
        return -1;
    }
    
    /* ==================== 修改点（原 → 现） ====================
     * 原: sscanf(input, "%[^:]:%s", cmd, arg);
     *     —— %[^:] 与 %s 都没有宽度限制，超长输入写穿 cmd/arg 缓冲区
     *     strcpy(cmd, input); 同样无长度限制
     *     且每次执行都打印 3 行调试信息，实际运行时刷屏
     * 现: 加最大宽度（63/127），用 snprintf 截断兜底，删除调试打印
     * 讲解: 键盘输入不可信，格式化解析必须限制写入长度。
     * ========================================================= */
    sscanf(input, "%63[^:]:%127s", cmd, arg);
    if (cmd[0] == '\0') {
        snprintf(cmd, sizeof(cmd), "%s", input);   /* 修改：原为 strcpy，防溢出 */
    }
    
    for (i = 0; i < cmd_count; i++) {
        if (strcmp(cmd, cmd_table[i].name) == 0) {
            if (cmd_table[i].handler == NULL) {
                return -1;
            }
            cmd_table[i].handler(arg);
            return 0;
        }
    }
    
    return -1;
}

void cmd_show_help(void)
{
    int i;
    
    printf("============= Available Commands =============\n");
    for (i = 0; i < cmd_count; i++) {
        printf("  %s\n", cmd_table[i].name);
    }
    printf("==============================================\n");
}
```

```c
 * 文件: static_pool.h
 * 功能: 静态内存池（替代 malloc/free，消除内存碎片）
 * 作者: [你的名字]
 * 版本: 1.0
 * =============================================================================
 * 
 * 【为什么需要静态内存池？】
 *   嵌入式系统长期运行下，频繁 malloc/free 会产生内存碎片，
 *   最终导致 malloc 失败。静态内存池预分配一块大内存，
 *   按固定大小块管理，无碎片、分配速度快。
```

```c
/**
 * =============================================================================
 * 文件: static_pool.c
 * 功能: 实现 static_pool.h 中声明的所有函数
 * =============================================================================
 * 
 * 【位图管理说明】
 *   used 数组是位图，每个比特表示一个块的占用状态：
 *   - bit=1: 已占用
 *   - bit=0: 空闲
 *   
 *   例如: used[0] 的 bit0 表示第0个块, bit1 表示第1个块 ...
 *   定位第 n 个块: byte = n/8, bit = n%8
 * 
 * 【分配算法】
 *   1. 遍历位图，找到第一个为0的bit
 *   2. 将其置1
 *   3. 返回该块的内存地址: pool + block_index * block_size
 * 
 * 【释放算法】
 *   1. 计算指针偏移: (ptr - pool) / block_size
 *   2. 将对应的bit置0
 * 
 * 【调用关系】
 *   业务代码
 *        ↓
 *   pool_alloc(pool)
 *        ↓
 *   遍历位图 → 找空闲块 → 置位 → 返回地址
 *   
 *   业务代码
 *        ↓
 *   pool_free(pool, ptr)
 *        ↓
 *   计算块索引 → 清除位 → 完成
 * 
 * 【线程安全】
 *   修改点：结构体内置 pthread_mutex_t 互斥锁，
 *   pool_alloc/pool_free/pool_stats/pool_reset 内部自动加锁，
 *   可直接在多线程环境中使用（原实现无锁，并发分配/释放会破坏位图）。
 * =============================================================================
 */

#include <stdlib.h>
#include <string.h>
#include <pthread.h>   /* 修改点（新增）：内存池互斥锁 */
#include "static_pool.h"
#include "logger.h"    /* 修改点（新增）：用日志宏替代裸 printf */

/* ==================== 私有辅助函数 ==================== */

/**
 * 将位图全置为0（全部空闲）
 */
static void pool_clear_used(STATIC_POOL *pool)
{
    int bytes = (pool->block_count + 7) / 8;
    memset(pool->used, 0, bytes);
}

/**
 * 检查位图某一位是否为1
 */
static int pool_is_used(STATIC_POOL *pool, int index)
{
    int byte = index / 8;
    int bit = index % 8;
    return (pool->used[byte] & (1 << bit)) ? 1 : 0;
}

/**
 * 设置位图某一位为1（标记已占用）
 */
static void pool_set_used(STATIC_POOL *pool, int index)
{
    int byte = index / 8;
    int bit = index % 8;
    pool->used[byte] |= (1 << bit);
}

/**
 * 设置位图某一位为0（标记空闲）
 */
static void pool_clear_used_bit(STATIC_POOL *pool, int index)
{
    int byte = index / 8;
    int bit = index % 8;
    pool->used[byte] &= ~(1 << bit);
}

/* ==================== 公共函数实现 ==================== */

int pool_init(STATIC_POOL *pool, int count, int size)
{
    if (pool == NULL || count <= 0 || size <= 0) {
        return -1;
    }
    
    /* 在堆上分配内存池 */
    pool->pool = (unsigned char *)malloc(count * size);
    if (pool->pool == NULL) {
        return -1;
    }
    
    /* 分配位图 */
    int used_bytes = (count + 7) / 8;
    pool->used = (unsigned char *)malloc(used_bytes);
    if (pool->used == NULL) {
        free(pool->pool);
        return -1;
    }
    
    pool->block_count = count;
    pool->block_size = size;
    
    /* 修改点（新增）：初始化互斥锁 */
    pthread_mutex_init(&pool->lock, NULL);
    
    /* 初始全部空闲 */
    pool_clear_used(pool);
    
    return 0;
}

void pool_destroy(STATIC_POOL *pool)
{
    if (pool == NULL) {
        return;
    }
    /* 修改点（新增）：销毁互斥锁（须在释放资源前） */
    pthread_mutex_destroy(&pool->lock);
    if (pool->pool != NULL) {
        free(pool->pool);
        pool->pool = NULL;
    }
    if (pool->used != NULL) {
        free(pool->used);
        pool->used = NULL;
    }
    pool->block_count = 0;
    pool->block_size = 0;
}

void *pool_alloc(STATIC_POOL *pool)
{
    int i;
    
    if (pool == NULL || pool->pool == NULL) {
        return NULL;
    }
    
    /* ==================== 修改点（原 → 现） ====================
     * 原: 直接遍历位图分配，无锁。
     *     主线程 pool_alloc() 与转发线程 pool_free() 并发时，
     *     “查位→置位”两步不是原子的，可能同一块被分配两次
     * 现: 加互斥锁保护位图操作
     * ========================================================= */
    pthread_mutex_lock(&pool->lock);
    
    /* 遍历位图找空闲块 */
    for (i = 0; i < pool->block_count; i++) {
        if (!pool_is_used(pool, i)) {
            pool_set_used(pool, i);
            pthread_mutex_unlock(&pool->lock);
            return pool->pool + i * pool->block_size;
        }
    }
    
    pthread_mutex_unlock(&pool->lock);
    return NULL;  /* 池已满 */
}

void pool_free(STATIC_POOL *pool, void *ptr)
{
    int index;
    unsigned char *p = (unsigned char *)ptr;
    
    if (pool == NULL || ptr == NULL || pool->pool == NULL) {
        LOG_DEBUG("pool_free: 无效参数");   /* 修改：printf → 日志宏 */
        return;
    }
    
    /* 修改点（新增）：加锁，避免与主线程 pool_alloc 并发破坏位图 */
    pthread_mutex_lock(&pool->lock);
    
    if (p < pool->pool || p >= pool->pool + pool->block_count * pool->block_size) {
        LOG_DEBUG("pool_free: ptr 不在池范围内");   /* 修改：printf → 日志宏 */
        pthread_mutex_unlock(&pool->lock);
        return;
    }
    
    index = (p - pool->pool) / pool->block_size;
    
    if (index < 0 || index >= pool->block_count) {
        LOG_DEBUG("pool_free: index 越界");   /* 修改：printf → 日志宏 */
        pthread_mutex_unlock(&pool->lock);
        return;
    }
    
    pool_clear_used_bit(pool, index);
    pthread_mutex_unlock(&pool->lock);   /* 修改点（新增）：解锁 */
}

void pool_stats(STATIC_POOL *pool, int *used, int *total, int *free)
{
    int i;
    int used_count = 0;
    
    if (pool == NULL) {
        if (used) *used = 0;
        if (total) *total = 0;
        if (free) *free = 0;
        return;
    }
    
    /* 修改点（新增）：加锁遍历位图 */
    pthread_mutex_lock(&pool->lock);
    
    for (i = 0; i < pool->block_count; i++) {
        if (pool_is_used(pool, i)) {
            used_count++;
        }
    }
    
    if (used) *used = used_count;
    if (total) *total = pool->block_count;
    if (free) *free = pool->block_count - used_count;
    
    pthread_mutex_unlock(&pool->lock);
}

void pool_reset(STATIC_POOL *pool)
{
    if (pool == NULL || pool->used == NULL) {
        return;
    }
    /* 修改点（新增）：加锁后清空位图 */
    pthread_mutex_lock(&pool->lock);
    pool_clear_used(pool);
    pthread_mutex_unlock(&pool->lock);
}
```
